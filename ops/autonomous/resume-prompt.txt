        ls | /usr/bin/grep '\.patch$' | while read -r p; do
          case "$p" in *.commits.patch|*.untracked-*) continue;; esac
          n="${p%.patch}"
          if   ls | /usr/bin/grep -qF -- "-$n."; then echo "decided  $n"
          elif [ -f "$n.commits.patch" ];       then echo "ORPHAN   $n  + UNPUSHED COMMITS"
          else                                       echo "ORPHAN   $n"; fi
        done
        ls | /usr/bin/grep '\.commits\.patch$' | while read -r c; do
          n="${c%.commits.patch}"
          ls | /usr/bin/grep -qF -- "-$n.commits." || echo "UNFILED COMMITS  $c"
        done
        ls | /usr/bin/grep -- '\.ignored-' | while read -r g; do
          case "$g" in *.bak|LANDED-as-*|SUPERSEDED-by-*|REJECTED-*|PARKED-*) continue;; esac

      before=$(git rev-parse HEAD)
      git commit -F /tmp/<you>-msg.txt > /tmp/<you>-commit.log 2>&1 &

        /usr/bin/grep -qE 'TESTS FAILED|commit refused' /tmp/<you>-commit.log && { echo REFUSED; break; }
